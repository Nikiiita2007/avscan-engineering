cmake_minimum_required(VERSION 3.24)
project(avscan_engineering VERSION 0.4.0 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
set(CMAKE_EXPORT_COMPILE_COMMANDS ON)
option(AV_OFFLINE "Build without real tl::expected (explicit local Result fallback)" OFF)
option(AV_ARCHIVES "Enable zlib and libarchive" ON)
option(AV_ASAN "AddressSanitizer" OFF)
option(AV_UBSAN "UndefinedBehaviorSanitizer (GCC/Clang only)" OFF)
option(AV_TIDY "Run clang-tidy during compile" OFF)
option(AV_FUZZ "Build libFuzzer targets (Clang only)" OFF)
option(AV_CAPI "Build the C API shared library (avscan_capi), its C test and install rules" OFF)
option(AV_HOST "Build the module host core (header-only libs/host) and its tests" ON)
if(AV_TIDY)
 find_program(TIDY clang-tidy REQUIRED)
 set(CMAKE_CXX_CLANG_TIDY "${TIDY}")
endif()
if(AV_UBSAN AND MSVC)
 message(FATAL_ERROR "MSVC UBSan unsupported; use Linux Clang/GCC")
endif()
if(MSVC)
 add_compile_options(/W4 /utf-8)
 if(AV_ASAN)
  add_compile_options(/fsanitize=address)
  add_link_options(/INCREMENTAL:NO)
 endif()
else()
 add_compile_options(-Wall -Wextra)
 if(AV_ASAN)
  add_compile_options(-fsanitize=address -fno-omit-frame-pointer)
  add_link_options(-fsanitize=address)
 endif()
 if(AV_UBSAN)
  add_compile_options(-fsanitize=undefined -fno-sanitize-recover=undefined)
  add_link_options(-fsanitize=undefined)
 endif()
endif()
if(AV_FUZZ)
 if(NOT CMAKE_CXX_COMPILER_ID MATCHES "Clang" OR MSVC)
  message(FATAL_ERROR "libFuzzer build requires Linux clang++ (not clang-cl)")
 endif()
endif()
find_package(Threads REQUIRED)
add_library(core STATIC libs/core/scanner.cpp libs/core/archive.cpp libs/core/report.cpp)
target_include_directories(core PUBLIC libs/core)
if(NOT AV_OFFLINE)
 find_package(tl-expected CONFIG REQUIRED)
 target_link_libraries(core PUBLIC tl::expected)
 target_compile_definitions(core PUBLIC AV_USE_TL_EXPECTED)
endif()
if(AV_ARCHIVES)
 find_package(ZLIB REQUIRED)
 find_package(LibArchive REQUIRED)
 target_compile_definitions(core PRIVATE AV_HAVE_ZLIB AV_HAVE_LIBARCHIVE)
 target_link_libraries(core PRIVATE ZLIB::ZLIB LibArchive::LibArchive)
endif()
# (monitor is defined after quarantine above: monitor -> core + quarantine + Threads)
add_library(quarantine STATIC libs/quarantine/quarantine.cpp libs/quarantine/secure.cpp)
target_include_directories(quarantine PUBLIC libs/quarantine)
target_link_libraries(quarantine PUBLIC core)
add_library(gui STATIC libs/gui/gui.cpp)
target_include_directories(gui PUBLIC libs/gui)
target_link_libraries(gui PUBLIC core monitor)
add_library(monitor STATIC libs/monitor/monitor.cpp libs/monitor/etw.cpp libs/monitor/service.cpp)
target_include_directories(monitor PUBLIC libs/monitor)
target_link_libraries(monitor PUBLIC core quarantine Threads::Threads)
if(WIN32)
 target_compile_definitions(core PUBLIC NOMINMAX WIN32_LEAN_AND_MEAN UNICODE _UNICODE _WIN32_WINNT=0x0A00)
 target_link_libraries(monitor PUBLIC advapi32 tdh shell32 ole32)
 target_link_libraries(quarantine PUBLIC bcrypt crypt32 advapi32)
 target_link_libraries(gui PRIVATE comdlg32 user32)
endif()
add_executable(avscan apps/main.cpp apps/selftest.cpp)
target_link_libraries(avscan PRIVATE core)
add_executable(avmonitor apps/monitor_main.cpp)
target_link_libraries(avmonitor PRIVATE monitor)
if(MINGW)
 target_link_options(avmonitor PRIVATE -municode)
endif()
add_executable(avquarantine apps/quarantine_main.cpp)
target_link_libraries(avquarantine PRIVATE quarantine)
add_executable(avgui apps/gui_main.cpp)
target_link_libraries(avgui PRIVATE gui)
add_executable(quality_tests tests/quality.cpp)
target_link_libraries(quality_tests PRIVATE core quarantine)
add_executable(config_events_tests tests/config_events_test.cpp)
target_include_directories(config_events_tests PRIVATE libs/monitor)
target_link_libraries(config_events_tests PRIVATE core)
enable_testing()
add_test(NAME core_selftest COMMAND avscan --selftest)
add_test(NAME quality COMMAND quality_tests ${CMAKE_SOURCE_DIR}/db/signatures.db ${CMAKE_SOURCE_DIR}/tests/corpus)
add_test(NAME config_events COMMAND config_events_tests)
foreach(t avscan avmonitor avgui)
 add_custom_command(TARGET ${t} POST_BUILD
  COMMAND ${CMAKE_COMMAND} -E make_directory $<TARGET_FILE_DIR:${t}>/db
  COMMAND ${CMAKE_COMMAND} -E copy_if_different ${CMAKE_SOURCE_DIR}/db/signatures.db $<TARGET_FILE_DIR:${t}>/db/signatures.db)
endforeach()
if(AV_FUZZ)
 add_library(core_fuzz STATIC libs/core/scanner.cpp libs/core/archive.cpp libs/core/report.cpp)
 target_link_libraries(core_fuzz PUBLIC core)
 target_compile_options(core_fuzz PRIVATE -fsanitize=fuzzer-no-link)
 if(AV_ARCHIVES)
  target_compile_definitions(core_fuzz PRIVATE AV_HAVE_ZLIB AV_HAVE_LIBARCHIVE)
  target_link_libraries(core_fuzz PRIVATE ZLIB::ZLIB LibArchive::LibArchive)
 endif()
 foreach(kind pe archive)
  add_executable(fuzz_${kind} fuzz/${kind}.cpp)
  target_link_libraries(fuzz_${kind} PRIVATE core_fuzz)
  target_compile_options(fuzz_${kind} PRIVATE -fsanitize=fuzzer-no-link)
  target_link_options(fuzz_${kind} PRIVATE -fsanitize=fuzzer)
 endforeach()
endif()

# C API: shared library with the core linked in statically (see docs/CAPI.md).
if(AV_CAPI)
 enable_language(C)
 include(GNUInstallDirs)
 include(CMakePackageConfigHelpers)
 set_target_properties(core PROPERTIES POSITION_INDEPENDENT_CODE ON)
 add_library(avscan_capi SHARED libs/capi/avscan_capi.cpp)
 target_include_directories(avscan_capi PUBLIC
  $<BUILD_INTERFACE:${CMAKE_CURRENT_SOURCE_DIR}/libs/capi>
  $<INSTALL_INTERFACE:${CMAKE_INSTALL_INCLUDEDIR}>)
 target_compile_definitions(avscan_capi PRIVATE AVSCAN_BUILD INTERFACE AVSCAN_SHARED)
 target_link_libraries(avscan_capi PRIVATE core)
 set_target_properties(avscan_capi PROPERTIES
  CXX_VISIBILITY_PRESET hidden
  VISIBILITY_INLINES_HIDDEN ON
  VERSION ${PROJECT_VERSION}
  SOVERSION 1
  EXPORT_NAME capi)
 add_library(avscan::capi ALIAS avscan_capi)
 add_executable(capi_test tests/capi_test.c)
 target_link_libraries(capi_test PRIVATE avscan_capi)
 add_test(NAME capi COMMAND capi_test ${CMAKE_SOURCE_DIR}/db/signatures.db ${CMAKE_SOURCE_DIR}/tests/corpus)
 install(TARGETS avscan_capi EXPORT avscanTargets
  RUNTIME DESTINATION ${CMAKE_INSTALL_BINDIR}
  LIBRARY DESTINATION ${CMAKE_INSTALL_LIBDIR}
  ARCHIVE DESTINATION ${CMAKE_INSTALL_LIBDIR}
  INCLUDES DESTINATION ${CMAKE_INSTALL_INCLUDEDIR})
 install(FILES libs/capi/avscan.h DESTINATION ${CMAKE_INSTALL_INCLUDEDIR})
 install(EXPORT avscanTargets NAMESPACE avscan:: FILE avscanTargets.cmake
  DESTINATION ${CMAKE_INSTALL_LIBDIR}/cmake/avscan)
 write_basic_package_version_file(${CMAKE_CURRENT_BINARY_DIR}/avscanConfigVersion.cmake
  VERSION ${PROJECT_VERSION} COMPATIBILITY SameMajorVersion)
 install(FILES cmake/avscanConfig.cmake ${CMAKE_CURRENT_BINARY_DIR}/avscanConfigVersion.cmake
  DESTINATION ${CMAKE_INSTALL_LIBDIR}/cmake/avscan)
endif()

# Module host core (header-only): JSON, manifests, registry, protocol, lifecycle. See docs/MODULES.md.
if(AV_HOST)
 add_library(avhost INTERFACE)
 target_include_directories(avhost INTERFACE libs/host)
 add_executable(host_tests tests/host_test.cpp)
 target_link_libraries(host_tests PRIVATE avhost)
 add_test(NAME host COMMAND host_tests ${CMAKE_SOURCE_DIR}/modules)
endif()
